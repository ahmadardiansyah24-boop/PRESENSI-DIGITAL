Skrip SQL DDL Supabase

Jalankan di SQL Editor Supabase untuk migrasi tabel dan websocket realtime.

-- 1. Buat Tabel Siswa (students)
CREATE TABLE IF NOT EXISTS public.students (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nisn TEXT UNIQUE NOT NULL,
    name TEXT NOT NULL,
    class_name TEXT NOT NULL,
    face_descriptor JSONB NOT NULL,
    photo_url TEXT,
    created_at TIMESTAMPTZ DEFAULT now()
);

-- 2. Buat Tabel Presensi (attendances)
CREATE TABLE IF NOT EXISTS public.attendances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES public.students(id) ON DELETE CASCADE,
    class_name TEXT NOT NULL,
    date DATE DEFAULT CURRENT_DATE NOT NULL,
    time TIME DEFAULT CURRENT_TIME NOT NULL,
    status TEXT DEFAULT 'Hadir' NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now(),
    CONSTRAINT unique_student_daily_attendance UNIQUE (student_id, date)
);

-- 3. Buat Tabel Surat Izin & Sakit (permissions)
CREATE TABLE IF NOT EXISTS public.permissions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES public.students(id) ON DELETE CASCADE,
    nisn TEXT NOT NULL,
    student_name TEXT NOT NULL,
    class_name TEXT NOT NULL,
    type TEXT NOT NULL DEFAULT 'Sakit', -- 'Sakit', 'Izin', 'Dispensasi'
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    reason TEXT NOT NULL,
    parent_name TEXT NOT NULL,
    parent_phone TEXT,
    attachment_url TEXT,
    doc_number TEXT,
    created_at TIMESTAMPTZ DEFAULT now()
);

-- 4. RLS & Kebijakan Keamanan
ALTER TABLE public.students ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.attendances ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.permissions ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Akses publik baca data siswa" ON public.students FOR SELECT TO anon, authenticated USING (true);
CREATE POLICY "Akses publik tambah siswa baru" ON public.students FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Akses publik perbarui siswa" ON public.students FOR UPDATE TO anon, authenticated USING (true);
CREATE POLICY "Akses publik hapus siswa" ON public.students FOR DELETE TO anon, authenticated USING (true);

CREATE POLICY "Akses publik baca presensi" ON public.attendances FOR SELECT TO anon, authenticated USING (true);
CREATE POLICY "Akses publik catat presensi" ON public.attendances FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Akses publik update presensi" ON public.attendances FOR UPDATE TO anon, authenticated USING (true);
CREATE POLICY "Akses publik hapus presensi" ON public.attendances FOR DELETE TO anon, authenticated USING (true);

CREATE POLICY "Akses publik baca izin" ON public.permissions FOR SELECT TO anon, authenticated USING (true);
CREATE POLICY "Akses publik tambah izin" ON public.permissions FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Akses publik update izin" ON public.permissions FOR UPDATE TO anon, authenticated USING (true);
CREATE POLICY "Akses publik hapus izin" ON public.permissions FOR DELETE TO anon, authenticated USING (true);

ALTER PUBLICATION supabase_realtime ADD TABLE public.attendances;
ALTER PUBLICATION supabase_realtime ADD TABLE public.students;
ALTER PUBLICATION supabase_realtime ADD TABLE public.permissions;
